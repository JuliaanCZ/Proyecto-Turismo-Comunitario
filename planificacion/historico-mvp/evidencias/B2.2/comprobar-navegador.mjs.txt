async function launch(name) {
  const port = await freePort();
  const profile = mkdtempSync(join(tmpdir(), `tc-b22-${name}-`));
  const proc = spawn(CHROME, ['--headless=new', `--remote-debugging-port=${port}`, `--user-data-dir=${profile}`,
    '--no-first-run', '--disable-gpu', '--window-size=1280,800', 'about:blank'], { stdio: 'ignore' });
  for (let i = 0; i < 60; i += 1) {
    try { await (await fetch(`http://127.0.0.1:${port}/json/version`)).json(); break; } catch { await sleep(200); }
  }
  const version = (await (await fetch(`http://127.0.0.1:${port}/json/version`)).json()).Browser;
  return { name, port, proc, profile, version };
}

async function openTab(browser) {
  const target = await (await fetch(`http://127.0.0.1:${browser.port}/json/new?about:blank`, { method: 'PUT' })).json();
  const ws = new WebSocket(target.webSocketDebuggerUrl);
  await new Promise((r) => ws.addEventListener('open', r, { once: true }));
  let nextId = 1;
  const pending = new Map();
  const listeners = [];
  ws.addEventListener('message', (event) => {
    const msg = JSON.parse(event.data);
    if (msg.id && pending.has(msg.id)) {
      const { resolve, reject } = pending.get(msg.id);
      pending.delete(msg.id);
      return msg.error ? reject(new Error(msg.error.message)) : resolve(msg.result);
    }
    for (const fn of [...listeners]) fn(msg);
    return undefined;
  });
  const send = (method, params = {}) => new Promise((resolve, reject) => {
    const id = nextId++;
    pending.set(id, { resolve, reject });
    ws.send(JSON.stringify({ id, method, params }));
  });
  const tab = { problems: [], send, name: browser.name };
  const requests = new Map();
  listeners.push((msg) => {
    const p = msg.params;
    if (msg.method === 'Log.entryAdded' && p.entry.level === 'error') tab.problems.push(`${p.entry.text} ${p.entry.url ?? ''}`);
    if (msg.method === 'Runtime.exceptionThrown') tab.problems.push(`excepción: ${p.exceptionDetails.text}`);
    if (msg.method === 'Network.responseReceived') {
      requests.set(p.requestId, p.response.url);
      if (p.response.status >= 400 && p.type !== 'Document') tab.problems.push(`HTTP ${p.response.status} ${p.response.url}`);
    }
    if (msg.method === 'Network.loadingFailed') tab.problems.push(`red falló: ${requests.get(p.requestId)} ${p.errorText}`);
  });
  for (const domain of ['Page', 'Runtime', 'Log', 'Network']) await send(`${domain}.enable`);
  tab.viewport = async (kind) => {
    const mobile = kind === 'movil';
    await send('Emulation.setDeviceMetricsOverride', mobile
      ? { width: 390, height: 844, deviceScaleFactor: 2, mobile: true }
      : { width: 1280, height: 800, deviceScaleFactor: 1, mobile: false });
    tab.kind = kind;
  };
  tab.evaluate = async (expression) => {
    const r = await send('Runtime.evaluate', { expression, returnByValue: true, awaitPromise: true });
    if (r.exceptionDetails) throw new Error(`${r.exceptionDetails.text} ${r.exceptionDetails.exception?.description ?? ''}`);
    return r.result.value;
  };
  tab.waitLoad = () => new Promise((resolve) => {
    const fn = (msg) => { if (msg.method === 'Page.loadEventFired') { listeners.splice(listeners.indexOf(fn), 1); resolve(); } };
    listeners.push(fn);
  });
  tab.go = async (path) => {
    const loaded = tab.waitLoad();
    await send('Page.navigate', { url: `${BASE}${path}` });
    await loaded;
    await sleep(350);
  };
  tab.path = () => tab.evaluate('location.pathname + location.search');
  tab.waitFor = async (expression, ms = 6000) => {
    const end = Date.now() + ms;
    while (Date.now() < end) { if (await tab.evaluate(expression)) return true; await sleep(100); }
    return false;
  };
  tab.submitForm = async (selector, values) => {
    const loaded = tab.waitLoad();
    await tab.evaluate(`(() => {
      const form = document.querySelector(${JSON.stringify(selector)});
      for (const [name, value] of Object.entries(${JSON.stringify(values)})) form.elements.namedItem(name).value = value;
      form.requestSubmit(); return true; })()`);
    await loaded;
    await sleep(400);
  };
  tab.shot = async (name) => {
    const { data } = await send('Page.captureScreenshot', { format: 'png' });
    writeFileSync(join(OUT, name), Buffer.from(data, 'base64'));
  };
  tab.key = async (key, code, vk, modifiers = 0) => {
    for (const type of ['keyDown', 'keyUp']) {
      await send('Input.dispatchKeyEvent', { type, key, code, windowsVirtualKeyCode: vk, modifiers });
    }
  };
  tab.close = () => ws.close();
  return tab;
}

const AUDIT = `(() => {
  const describe = (el) => (el.tagName.toLowerCase() + (el.id ? '#' + el.id : '') + (el.className && typeof el.className === 'string' ? '.' + el.className.trim().split(/\\s+/).join('.') : '')).slice(0, 70);
  const vw = document.documentElement.clientWidth;
  const wide = [...document.querySelectorAll('body *')].filter((el) => { const r = el.getBoundingClientRect(); return r.width > 0 && (r.right > vw + 1 || r.left < -1); }).slice(0, 5).map(describe);
  const textSizes = [...document.querySelectorAll('p, li, label, a, button, dd, dt, span, h1, h2, h3, textarea, input, select')]
    .filter((el) => el.offsetParent !== null && (el.textContent || el.value || '').trim()).map((el) => parseFloat(getComputedStyle(el).fontSize));
  return {
    title: document.title, lang: document.documentElement.lang, h1: document.querySelectorAll('h1').length,
    main: !!document.querySelector('main, [role=main]'), skip: !!document.querySelector('a.saltar, a[href="#contenido"], a[href^="#main"], a[href^="#"][class*="salt"]'),
    scrollW: document.documentElement.scrollWidth, clientW: vw, overflowX: document.documentElement.scrollWidth - vw, wide,

    imgs: [...document.images].map((i) => ({ ok: i.complete && i.naturalWidth > 0, alt: i.alt, src: i.getAttribute('src'), w: Math.round(i.getBoundingClientRect().width) })),
    minFont: textSizes.length ? Math.min(...textSizes) : null,

    h1text: document.querySelector('h1')?.textContent.trim().slice(0, 60) ?? null,
  };
})()`;

const FOCUS_INFO = `(() => {
  const el = document.activeElement;
  if (!el || el === document.body) return null;
  const cs = getComputedStyle(el);
  const r = el.getBoundingClientRect();
  const outline = cs.outlineStyle !== 'none' && parseFloat(cs.outlineWidth) > 0;
  const shadow = cs.boxShadow !== 'none';
  const native = /^(date|datetime-local|time|month|week)$/.test(el.type || '') && !el.matches(':focus');
  return { native, fv: el.matches(':focus-visible'), tag: el.tagName.toLowerCase(), id: el.id || null, name: (el.getAttribute('aria-label') || el.textContent || el.getAttribute('name') || el.getAttribute('href') || '').trim().replace(/\\s+/g, ' ').slice(0, 40),
    visible: outline || shadow, outline: outline ? cs.outlineWidth + ' ' + cs.outlineStyle : 'ninguno', inView: r.bottom > 0 && r.top < innerHeight && r.right > 0 && r.left < innerWidth };
})()`;

async function tabThrough(tab, label, max = 45) {
  await tab.evaluate('document.activeElement && document.activeElement.blur(); window.scrollTo(0, 0); true');
  const stops = [];
  for (let i = 0; i < max; i += 1) {
    await tab.key('Tab', 'Tab', 9);
    await sleep(25);
    const f = await tab.evaluate(FOCUS_INFO);
    if (!f) break;
    const signature = `${f.tag}|${f.id}|${f.name}`;
    if (stops.length && signature === stops[0].signature) break;
    // Un campo de fecha y hora tiene varias paradas internas (día, mes, año…): se cuenta una vez por campo.
    if (stops.length && signature === stops.at(-1).signature) {
      const last = stops.at(-1);
      last.internas = (last.internas ?? 1) + 1;
      if (f.native) { last.native = true; await tab.shot(`${tab.kind}-foco-selector-fecha.png`); }
      else if (!f.visible) { last.visible = false; last.fv = f.fv; last.outline = f.outline; }
      continue;
    }
    stops.push({ ...f, signature });
    if (f.native) await tab.shot(`${tab.kind}-foco-selector-fecha.png`);
  }
  const nativeStops = stops.filter((s) => s.native);
  if (nativeStops.length) info(`${label}: parada nativa`, `el icono del selector de fecha de «${nativeStops[0].name}» recibe el foco y el navegador dibuja su propio anillo (ver captura *-foco-selector-fecha.png); el CSS del proyecto no lo controla`);
  const hidden = stops.filter((s) => !s.visible && !s.native);
  const offscreen = stops.filter((s) => !s.inView);
  check(stops.length >= 3, `${label}: se puede recorrer con Tab`, `${stops.length} paradas; primera: ${stops[0]?.name}`);
  check(hidden.length === 0, `${label}: todas las paradas muestran foco visible`, hidden.length ? `sin foco: ${hidden.map((h) => `${h.tag} «${h.name}» (focus-visible=${h.fv}, outline=${h.outline})`).join('; ')}` : `${stops.length}/${stops.length}`);
  check(offscreen.length === 0, `${label}: el foco no queda fuera de la ventana`, offscreen.map((h) => h.name).join('; '));
  return stops;
}

async function audit(tab, path, slug, { expectMenu, expectImages = false, shot = true } = {}) {
  await tab.go(path);
  const a = await tab.evaluate(AUDIT);
  const where = `${tab.kind} ${path}`;
  check(a.title.length > 3 && a.h1 === 1, `${where}: <title> propio y un único <h1>`, `«${a.title}» · h1=${a.h1} «${a.h1text}»`);
  check(a.overflowX <= 0, `${where}: sin desplazamiento horizontal`, `scrollWidth ${a.scrollW} / ancho ${a.clientW}${a.wide.length ? ` · anchos: ${a.wide.join(', ')}` : ''}`);
  check(a.viewportMeta && /^es/.test(a.lang), `${where}: viewport y lang="es-EC"`, `lang=${a.lang}`);
  check(a.minFont === null || a.minFont >= 14, `${where}: texto legible (mínimo ${a.minFont}px)`);
  if (expectMenu) check(JSON.stringify(a.menu) === JSON.stringify(expectMenu), `${where}: menú esperado`, a.menu.join(' '));
  if (expectImages) check(a.imgs.length > 0 && a.imgs.every((i) => i.ok && i.alt.trim()), `${where}: imágenes cargadas y con texto alternativo`, `${a.imgs.length} imagen(es)`);
  if (shot) await tab.shot(`${tab.kind}-${slug}.png`);
  return a;
}

  info('Duración de esta comprobación AUTOMÁTICA', `${Math.round((Date.now() - started) / 1000)} s (no es el tiempo de un ensayo humano)`);
  say(`\nResultado: ${passes} comprobaciones pasan, ${failures} fallan.`);
  exit = failures ? 1 : 0;
} catch (error) {
  say(`ERROR inesperado: ${error.stack}`);
  exit = 1;
} finally {
  for (const b of browsers) { b.proc.kill(); }
  server.kill();
  await sleep(800);
  for (const dir of [WORK, ...browsers.map((b) => b.profile)]) { try { rmSync(dir, { recursive: true, force: true, maxRetries: 5 }); } catch { /* perfil aún en uso */ } }
}
process.exit(exit);
