// Comprobación de B0.4 en un Chrome real (headless) por el protocolo DevTools. Sin dependencias.
// Uso: node comprobar-navegador.mjs <carpeta-de-salida> [url-base]
import { spawn } from 'node:child_process';
import { mkdirSync, mkdtempSync, writeFileSync } from 'node:fs';
import { tmpdir } from 'node:os';
import { join } from 'node:path';

const OUT = process.argv[2];
const BASE = process.argv[3] ?? 'http://localhost:3000';
const CHROME = process.env.CHROME ?? 'C:\\Program Files\\Google\\Chrome\\Application\\chrome.exe';
const PORT = 9333;
mkdirSync(OUT, { recursive: true });
const log = [];
const say = (line) => { console.log(line); log.push(line); };
const sleep = (ms) => new Promise((r) => setTimeout(r, ms));

const profile = mkdtempSync(join(tmpdir(), 'tc-chrome-'));
const chrome = spawn(CHROME, [
  '--headless=new', `--remote-debugging-port=${PORT}`, `--user-data-dir=${profile}`,
  '--no-first-run', '--disable-gpu', '--hide-scrollbars', 'about:blank',
], { stdio: 'ignore' });

async function connect() {
  for (let i = 0; i < 50; i++) {
    try {
      const list = await (await fetch(`http://127.0.0.1:${PORT}/json/list`)).json();
      const page = list.find((t) => t.type === 'page');
      if (page) return page.webSocketDebuggerUrl;
    } catch { /* aún arrancando */ }
    await sleep(200);
  }
  throw new Error('Chrome no respondió');
}

const ws = new WebSocket(await connect());
await new Promise((r) => ws.addEventListener('open', r, { once: true }));
let nextId = 1;
const pending = new Map();
const listeners = [];
ws.addEventListener('message', (event) => {
  const msg = JSON.parse(event.data);
  if (msg.id && pending.has(msg.id)) {
    const { resolve, reject } = pending.get(msg.id);
    pending.delete(msg.id);
    return msg.error ? reject(new Error(`${msg.error.message}`)) : resolve(msg.result);
  }
  for (const fn of listeners) fn(msg);
  return undefined;
});
const send = (method, params = {}) => new Promise((resolve, reject) => {
  const id = nextId++;
  pending.set(id, { resolve, reject });
  ws.send(JSON.stringify({ id, method, params }));
});
const evaluate = async (expression) => {
  const r = await send('Runtime.evaluate', { expression, returnByValue: true, awaitPromise: true });
  if (r.exceptionDetails) throw new Error(r.exceptionDetails.text + ' ' + (r.exceptionDetails.exception?.description ?? ''));
  return r.result.value;
};

const problems = [];
const requests = new Map();
listeners.push((msg) => {
  if (msg.method === 'Log.entryAdded' && ['error', 'warning'].includes(msg.params.entry.level)) {
    problems.push(`${msg.params.entry.level}: ${msg.params.entry.text} ${msg.params.entry.url ?? ''}`);
  }
  if (msg.method === 'Runtime.exceptionThrown') problems.push(`excepción: ${msg.params.exceptionDetails.text}`);
  if (msg.method === 'Runtime.consoleAPICalled' && ['error', 'warning'].includes(msg.params.type)) {
    problems.push(`consola.${msg.params.type}: ${msg.params.args.map((a) => a.value ?? a.description).join(' ')}`);
  }
  if (msg.method === 'Network.requestWillBeSent') requests.set(msg.params.requestId, msg.params.request.url);
  if (msg.method === 'Network.responseReceived') {
    const { url, status, mimeType } = msg.params.response;
    requests.set(msg.params.requestId, `${status} ${mimeType} ${url}`);
  }
  if (msg.method === 'Network.loadingFailed') problems.push(`red falló: ${requests.get(msg.params.requestId)} ${msg.params.errorText}`);
});

for (const domain of ['Page', 'Runtime', 'Log', 'Network']) await send(`${domain}.enable`);

async function go(path) {
  const loaded = new Promise((resolve) => {
    const fn = (msg) => { if (msg.method === 'Page.loadEventFired') { listeners.splice(listeners.indexOf(fn), 1); resolve(); } };
    listeners.push(fn);
  });
  await send('Page.navigate', { url: `${BASE}${path}` });
  await loaded;
  await sleep(400);
}
async function shot(name, clip) {
  const params = { format: 'png', captureBeyondViewport: true };
  if (clip) params.clip = { ...clip, scale: 1 };
  const { data } = await send('Page.captureScreenshot', params);
  writeFileSync(join(OUT, name), Buffer.from(data, 'base64'));
  say(`  captura: ${name}`);
}
async function fullShot(name) {
  const { cssContentSize } = await send('Page.getLayoutMetrics');
  await shot(name, { x: 0, y: 0, width: cssContentSize.width, height: cssContentSize.height });
}
async function key(k, code, vk) {
  for (const type of ['keyDown', 'keyUp']) {
    await send('Input.dispatchKeyEvent', { type, key: k, code, windowsVirtualKeyCode: vk, nativeVirtualKeyCode: vk });
  }
}
const tab = () => key('Tab', 'Tab', 9);

// ---------- Móvil 375 px ----------
say(`Chrome ${(await (await fetch(`http://127.0.0.1:${PORT}/json/version`)).json()).Browser}`);
say('== Móvil 375×812 (/dev/diseno)');
await send('Emulation.setDeviceMetricsOverride', { width: 375, height: 812, deviceScaleFactor: 1, mobile: true });
await go('/dev/diseno');

say(`módulo cargado y formulario enlazado: ${await evaluate("document.querySelector('form[data-schema]').noValidate")} (noValidate=true significa que form-validate.js corrió)`);
const layout = await evaluate(`(() => {
  const de = document.documentElement;
  const interactive = [...document.querySelectorAll('a, button, input, select, textarea')];
  const small = interactive.filter((e) => e.getBoundingClientRect().height > 0 && e.getBoundingClientRect().height < 43.5)
    .map((e) => e.tagName + '.' + e.className + ' ' + Math.round(e.getBoundingClientRect().height) + 'px');
  const cs = (sel, prop) => getComputedStyle(document.querySelector(sel))[prop];
  return {
    anchoVentana: window.innerWidth, anchoContenido: de.scrollWidth,
    desbordaHorizontal: de.scrollWidth > window.innerWidth,
    textoBase: cs('body', 'fontSize'), h1: cs('h1', 'fontSize'), nombreOferta: cs('.oferta__nombre', 'fontSize'),
    familiaTitulo: cs('h1', 'fontFamily').slice(0, 40), familiaTexto: cs('body', 'fontFamily').slice(0, 30),
    controlesBajoDe44px: small,
    viewportMeta: document.querySelector('meta[name=viewport]').content,
    lang: de.lang, tituloPagina: document.title, h1Cantidad: document.querySelectorAll('h1').length,
    mainId: document.querySelector('main').id,
    etiquetasSinInput: [...document.querySelectorAll('label')].filter((l) => !document.getElementById(l.htmlFor)).length,
  };
})()`);
say(`  ${JSON.stringify(layout)}`);
await fullShot('diseno-movil.png');

say('== Enlace «Saltar al contenido» y foco por teclado');
await tab();
const skip = await evaluate(`(() => { const e = document.activeElement; const r = e.getBoundingClientRect();
  const cs = getComputedStyle(e);
  return { texto: e.textContent, esSaltar: e.classList.contains('saltar'), visible: r.top >= 0 && r.bottom > 0, top: Math.round(r.top),
    focusVisible: e.matches(':focus-visible'), contorno: cs.outlineStyle + ' ' + cs.outlineWidth + ' ' + cs.outlineColor }; })()`);
say(`  primer Tab: ${JSON.stringify(skip)}`);
await shot('foco-saltar.png', { x: 0, y: 0, width: 375, height: 140 });
await key('Enter', 'Enter', 13);
await sleep(200);
say(`  tras Enter el foco pasa a: ${await evaluate('document.activeElement.id || document.activeElement.tagName')}`);

// Recorre con Tab hasta el primer botón del contenido y mide su anillo.
let focused;
for (let i = 0; i < 12; i++) {
  await tab();
  focused = await evaluate(`(() => { const e = document.activeElement; const cs = getComputedStyle(e);
    return { tag: e.tagName, texto: (e.textContent || e.name || '').trim().slice(0, 30), focusVisible: e.matches(':focus-visible'),
      outline: cs.outlineStyle + ' ' + cs.outlineWidth + ' ' + cs.outlineColor, offset: cs.outlineOffset }; })()`);
  if (focused.tag === 'BUTTON' && /Publicar/.test(focused.texto)) break;
}
say(`  botón con foco: ${JSON.stringify(focused)}`);
const rect = await evaluate(`(() => { const r = document.activeElement.getBoundingClientRect(); return { x: Math.max(0, r.x - 12), y: r.y + window.scrollY - 12, width: r.width + 24, height: r.height + 24 }; })()`);
await shot('foco-boton.png', rect);

const campos = ['nombre', 'correo', 'cupo'];
for (const campo of campos) {
  await evaluate(`document.getElementById('campo-${campo}').focus()`);
  const f = await evaluate(`(() => { const e = document.activeElement; const cs = getComputedStyle(e);
    return { id: e.id, focusVisible: e.matches(':focus-visible'), outline: cs.outlineStyle + ' ' + cs.outlineWidth + ' ' + cs.outlineColor }; })()`);
  say(`  foco en campo: ${JSON.stringify(f)}`);
}

say('== Validación en el navegador (mismo validate.js)');
await evaluate("document.getElementById('campo-nombre').blur()");
await evaluate("document.querySelector('form[data-schema] button[type=submit]').click()");
await sleep(300);
const errores = () => evaluate(`(() => Object.fromEntries([...document.querySelectorAll('[data-error-for]')].map((e) => [e.dataset.errorFor, e.textContent])))()`);
say(`  envío vacío → url sin cambiar: ${(await evaluate('location.pathname')) === '/dev/diseno'}; foco: ${await evaluate('document.activeElement.id')}`);
say(`  errores: ${JSON.stringify(await errores())}`);
say(`  aria-invalid en nombre: ${await evaluate("document.getElementById('campo-nombre').getAttribute('aria-invalid')")}; wrapper con error: ${await evaluate("document.getElementById('campo-nombre').closest('.campo').classList.contains('campo--error')")}`);
await evaluate("document.querySelector('form[data-schema]').scrollIntoView()");
const formRect = await evaluate(`(() => { const r = document.querySelector('form[data-schema]').getBoundingClientRect(); return { x: 0, y: r.y + window.scrollY - 16, width: 375, height: r.height + 32 }; })()`);
await shot('errores-formulario.png', formRect);

await evaluate("document.getElementById('campo-correo').focus()");
await send('Input.insertText', { text: 'no-es-correo' });
await evaluate("document.getElementById('campo-cupo').focus()");
say(`  correo inválido al salir del campo: ${JSON.stringify((await errores()).correo)}`);
await evaluate("document.getElementById('campo-correo').focus()");
await send('Input.insertText', { text: '@ejemplo.ec' });
say(`  al completar el correo con @ejemplo.ec el error desaparece (se revalida al escribir): ${JSON.stringify((await errores()).correo)}`);

await evaluate("document.getElementById('campo-nombre').focus()");
await send('Input.insertText', { text: 'Ana' });
await evaluate("document.getElementById('campo-cupo').focus()");
await send('Input.insertText', { text: '12' });
await evaluate("document.getElementById('campo-correo').value = ''; document.getElementById('campo-correo').focus()");
await send('Input.insertText', { text: 'ana@ejemplo.ec' });
const navegado = new Promise((resolve) => {
  const fn = (msg) => { if (msg.method === 'Page.loadEventFired') { listeners.splice(listeners.indexOf(fn), 1); resolve(); } };
  listeners.push(fn);
});
await evaluate("document.querySelector('form[data-schema] button[type=submit]').click()");
await navegado;
await sleep(300);
say(`  envío válido → POST real del navegador con su Origin: mensaje = ${JSON.stringify(await evaluate("document.querySelector('.mensaje--exito')?.textContent.trim() ?? null"))}`);

// ---------- Escritorio ----------
say('== Escritorio 1280×800');
await send('Emulation.setDeviceMetricsOverride', { width: 1280, height: 800, deviceScaleFactor: 1, mobile: false });
await go('/dev/diseno');
say(`  desborda horizontal: ${await evaluate('document.documentElement.scrollWidth > window.innerWidth')}`);
await shot('diseno-escritorio.png', { x: 0, y: 0, width: 1280, height: 800 });
await go('/');
say(`  portada: título "${await evaluate('document.title')}", h1 "${await evaluate("document.querySelector('h1').textContent")}"`);
await shot('portada-escritorio.png', { x: 0, y: 0, width: 1280, height: 420 });
await go('/ruta-que-no-existe');
say(`  404: h1 "${await evaluate("document.querySelector('h1').textContent")}"`);

say('== Problemas de consola, CSP o red durante toda la sesión (se excluye el 404 de la ruta inexistente, pedido a propósito)');
const relevantes = problems.filter((p) => !p.includes('/ruta-que-no-existe'));
say(relevantes.length ? relevantes.map((p) => `  ${p}`).join('\n') : '  ninguno');
say('== Peticiones de /js');
for (const value of requests.values()) if (/\/js\/|\/css\//.test(value) && /^\d/.test(value)) say(`  ${value}`);

writeFileSync(join(OUT, 'navegador-comprobaciones.txt'), `${log.join('\n')}\n`);
ws.close();
chrome.kill();
await sleep(500);
process.exit(relevantes.length ? 1 : 0);
